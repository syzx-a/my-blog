"categories" }}{{ $cats = . }}{{ end -}} {{- with index . "series" }}{{ $sers = . }}{{ end -}} {{- with index . "sections" }}{{ $secs = . }}{{ end -}} {{- end -}} {{- $out := slice -}} {{- range $p := site.RegularPages -}} {{- $hit := false -}} {{- with $p.Params.categories -}} {{- range . }}{{ if in $cats . }}{{ $hit = true }}{{ end }}{{ end -}} {{- end -}} {{- with $p.Params.series -}} {{- range . }}{{ if in $sers . }}{{ $hit = true }}{{ end }}{{ end -}} {{- end -}} {{- if in $secs $p.Section }}{{ $hit = true }}{{ end -}} {{- if $hit }}{{ $out = $out | append $p }}{{ end -}} {{- end -}} {{- return $out -}}
